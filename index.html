Google Apps Script Integration Guide

Paste the following code into your Google Apps Script editor (Code.gs) to enable two-way sync between this web portal and Google Sheets: 

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([
    data.timestamp || new Date().toLocaleString('en-US'),
    data.type || '',
    data.name || '',
    data.country || '',
    data.school || '',
    data.position || '',
    data.session || data.week || '',
    data.content || '',
    data.reflection || '',
    data.gender || '',     // J
    data.email || '',      // K
    data.phone || '',      // L
    data.referrer || '',   // M
    data.consent || ''     // N
  ]);
  return ContentService.createTextOutput(JSON.stringify({status: 'success'})).setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  var result = [];
  for (var i = 1; i < rows.length; i++) {
    var row = rows[i];
    if (row[0] || row[2]) {
      result.push({
        timestamp: row[0],
        type: row[1],
        name: row[2],
        country: row[3],
        school: row[4],
        position: row[5],
        session: row[6],
        week: row[6],
        content: row[7],
        reflection: row[8],
        gender: row[9] || '',
        email: row[10] || '',
        phone: row[11] || '',
        referrer: row[12] || '',
        consent: row[13] || ''
      });
    }
  }
  return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
}
