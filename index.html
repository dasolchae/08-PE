Google Apps Script Integration & Two-Way Sync Guide

To allow the web app to load and accumulate data directly from Google Sheets (Two-way sync), update your Google Apps Script (`Code.gs`) with both doPost and doGet functions: 

// Google Apps Script (Code.gs)
function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([data.timestamp, data.type, data.name, data.country, data.school, data.position || '', data.session || data.week || '', data.content || '', data.reflection || '']);
  return ContentService.createTextOutput(JSON.stringify({status: 'success'})).setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  var result = [];
  for (var i = 1; i < rows.length; i++) {
    var row = rows[i];
    if (row[0]) {
      result.push({
        timestamp: row[0],
        type: row[1],
        name: row[2],
        country: row[3],
        school: row[4],
        position: row[5],
        session: row[6],
        week: row[6],
        content: row[7],
        reflection: row[8]
      });
    }
  }
  return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
}
